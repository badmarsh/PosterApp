\block{The ATLAS Hadronic Tile Calorimeter (TileCal)}{
    \tightitems
        \item The \textbf{ATLAS} detector is a multi-purpose particle physics experiment at the CERN Large Hadron Collider (LHC). The High-Luminosity LHC (HL-LHC) upgrade will drastically increase the collision rate, delivering an unprecedented integrated luminosity of up to 4000\,fb$^{-1}$.
        \item The \textbf{Tile Calorimeter} is a hadronic calorimeter composed of wedge-shaped modules arranged in a Long Barrel (LB) and two Extended Barrels (EB). It consists of 256 modules containing 5182 calorimetric cells, utilizing scintillating tiles and wavelength shifting fibers to transfer light to photomultiplier tubes (PMTs). Most cells are read out by two PMTs, except for special E1--E4 gap/crack cells read out by one PMT.
    \end{itemize}
    \vspace{-0.1cm}
    \begin{center}
        \begin{minipage}{0.6\linewidth}
            \includegraphics[width=\linewidth]{assets/plot1.pdf}
        \end{minipage}
        \hfill
        \begin{minipage}{0.37\linewidth}
            \includegraphics[width=\linewidth]{assets/plot2.pdf}
        \end{minipage}
    \end{center}
    \vspace{-0.1cm}
    {\small \captiontext{Figure 1: ATLAS Tile Calorimeter (left) and a wedge-shaped module slice (right) [1].}}
    \vspace{-0.1cm}
    \tightitems
        \item Components located in the highest radiation regions of TileCal, particularly the LB LVPS and readout electronics close to the detector gap regions, must tolerate elevated \textcolor{tid}{TID} and \textcolor{niel}{NIEL} levels.
    \end{itemize}
    \vspace{-0.1cm}
    \begin{center}
        \includegraphics[width=0.9\linewidth]{assets/fig_architecture.png}
    \end{center}
    \vspace{-0.3cm}
    {\small \captiontext{Figure 2: Tile HL-LHC On-detector architecture.}}
    \vspace{-0.1cm}
    \tightitems
        \item \textbf{1792 $\times$ Low Voltage power supply bricks (LVPS)}: Utilize switching DC-DC converters at 300\,kHz to step down a bulk 200\,V input to distribute stable power to the on-detector readout electronics.
        \item \textbf{$\sim$1000 $\times$ new PMTs}: Hamamatsu R11187, Quantum Efficiency $>15\%$.
        \item \textbf{9852 $\times$ Active Dividers}: Provide stable high voltage division for the PMTs to minimize gain variations at high event rates.
        \item \textbf{9852 $\times$ FENICS cards}: PMT pulse shaping with bi-gain amplification (1:40). Current integrator for luminosity measurements and Cs calibration.
        \item \textbf{896 $\times$ MainBoards (MB)}: Control and configuration, FENICS signal digitization, 2 $\times$ 12-bit ADCs @ 40\,Msps, and 1 $\times$ 16-bit ADC for integrator.
        \item \textbf{896 $\times$ DaughterBoards (DB)}: Central communication hub of each module. They transmit $\sim$35\,Tbps of physics data (digitized PMT signals) to off-detector systems via 3584 optical uplinks (9.6\,Gbps), while receiving configuration and timing commands via 1792 downlinks (4.8\,Gbps).
    \end{itemize}
}

\block{FENICS \& Active Divider Radiation Qualification}{
    \textbf{FENICS Boards}
    \tightitems
        \item Over 40 FENICS boards (plus pre-production tracking) were irradiated with X-rays and neutrons to test for \textcolor{tid}{TID} and \textcolor{niel}{NIEL} limits (up to 51\,Gy and $4.7 \times 10^{12}$\,n/cm$^2$).
        \item Fast channels showed no performance degradation. Slow channels exhibited a reasonable and fully acceptable noise increase, officially certifying all active Commercial Off-The-Shelf (COTS) components for production.
    \end{itemize}
    \vspace{-0.2cm}
    \begin{center}
        \includegraphics[width=0.9\linewidth]{assets/fig_fenics_test.png}
    \end{center}
    \vspace{-0.4cm}
    {\small \captiontext{Figure 5: FENICS channel noise level over burn-in time.}}

    \vspace{0.2cm}
    \textbf{Active Dividers}
    \tightitems
        \item Qualified at 424\,Gy \textcolor{tid}{TID}, $1.37 \times 10^{13}$\,n/cm$^2$ \textcolor{niel}{NIEL}, and $2.97 \times 10^{12}$\,had/cm$^2$ \textcolor{see}{SEE}.
        \item Extensive proton, gamma, and neutron beam tests: \textbf{No \textcolor{see}{SEE}s were observed during irradiation}.
        \item Relative PMT gain decrease due to \textcolor{tid}{TID} effects (at 480\,Gy) remained strictly within tolerances at just $-1.92 \pm 0.20\%$.
    \end{itemize}

\block{MB Radiation Qualification}{
    \tightitems
        \item All components passed \textcolor{tid}{TID} (up to 640\,Gy) and \textcolor{niel}{NIEL} (up to $1.4 \times 10^{13}$\,n/cm$^2$) tests with no performance degradation.
        \item Critical components (Fast ADCs, FPGAs) showed zero \textcolor{seu}{SEU} effects up to $10^{13}$\,p/cm$^2$ during 800\,MeV proton testing. Point-of-Load regulators successfully passed stringent testing at 570\,Gy \textcolor{tid}{TID} and $1.2 \times 10^{13}$\,n/cm$^2$ \textcolor{niel}{NIEL}.
    \end{itemize}
}

\block{LVPS Component Radiation Testing}{
    \textbf{LVPS Design}
    \tightitems
        \item 8 identical bricks for LB modules and 6 bricks for EB modules inside a box convert the 200\,V input to a 10\,V output at a nominal power of 23\,W. Each brick powers an independent side of the readout electronics, preserving the TileCal dual-PMT cell redundancy.
        \item Replacing original STB57N65M5 power MOSFETs with SIHFS9N60A reduced gate-source capacitance and switching losses. Efficiency improved from 58\% to 72\%, meeting the 77\,kW TileCal cooling system capacity.
    \end{itemize}
    \vspace{1.0cm}
    \textbf{Radiation Testing}
    \tightitems
        \item SIHFS9N60A MOSFET (main high-power switching transistor): 2.5\,V shift in gate-source threshold observed after 200\,Gy of \textcolor{tid}{TID}, minimal deviation from \textcolor{niel}{NIEL}.
        \item SI8920 Isolation Amplifier (safe analog signal isolation for over-voltage protection): Initial tests showed voltage fluctuations at 150\,mV, but performance is completely stable below 100\,mV. It has been selected for final production with circuits restricting the input voltage.
        \item LT1681 Controller (feedback loop \& switching pulse generator) \& LT3080 Regulator (maintains stable local voltages): \textcolor{tid}{TID} tests at 502\,Gy/h showed minor drifts ($\leq$ 5\%) in output voltages, within acceptable limits.
        % \item LTC6241 Op-Amp (signal conditioning): Single Event Transients (SETs) had small amplitudes ($\approx$50\,mV) and short durations ($\approx$100\,ns) posing no effect on brick performance.
    \end{itemize}

\block{Conclusions}{
    \tightitems
        \item The optimized power distribution scheme improves readout reliability and power efficiency, and mitigates failure impact by utilizing identical and independent LVPS bricks.
        \item All critical components for the upgrade of the TileCal on-detector electronics successfully passed \textcolor{tid}{TID}, \textcolor{niel}{NIEL}, and \textcolor{see}{SEE} qualification, meeting the ATLAS radiation requirements for production.
    \end{itemize}
}
